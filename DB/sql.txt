CREATE DATABASE `BancoDados` /*!40100 DEFAULT CHARACTER SET utf8 */;
USE `BancoDados`;

CREATE TABLE `INFRACAO` (
  `ID_INFRACAO` int(11) NOT NULL AUTO_INCREMENT,
  `NOME_MOTORISTA` varchar(255) NOT NULL,
  `PLACA_VEICULO` varchar(10) NOT NULL,
  `NUMERO_RASTREADOR` varchar(40) NOT NULL,
  `DATA` date NOT NULL,
  `VELOCIDADE` double NOT NULL,
  `LIMITE_VELOCIDADE` double NOT NULL,
  PRIMARY KEY (`ID_INFRACAO`)
) ENGINE=InnoDB AUTO_INCREMENT=6 DEFAULT CHARSET=utf8;


CREATE TABLE `USUARIO` (
  `ID_USUARIO` int(11) NOT NULL AUTO_INCREMENT,
  `LOGIN` varchar(20) NOT NULL,
  `SENHA` varchar(20) NOT NULL,
  `ATIVO` smallint(6) NOT NULL,
  PRIMARY KEY (`ID_USUARIO`)
) ENGINE=InnoDB AUTO_INCREMENT=5 DEFAULT CHARSET=utf8;
